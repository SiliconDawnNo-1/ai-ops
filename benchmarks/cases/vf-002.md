## 场景
每次套件迭代交付后，与 vf-001 一起跑。

## 质量门槛
所有 case 的 frontmatter 含 id/dimension/type/status；judged case 必须指向存在的 rubric 文件；所有实验卡 status 在合法枚举内。

## 断言
python3 - <<'EOF'
import glob, os, sys
sys.path.insert(0, os.path.expanduser('~/Desktop/code/ai-ops/benchmarks/lib'))
from group_scores import parse_frontmatter, load_experiments
root = os.path.expanduser('~/Desktop/code/ai-ops')
ok = True
for f in sorted(glob.glob(root + '/benchmarks/cases/*.md')):
    fm = parse_frontmatter(f)
    if not fm or not all(fm.get(k) for k in ('id', 'dimension', 'type', 'status')):
        print('bad case frontmatter:', f); ok = False
    elif fm.get('type') == 'judged' and not os.path.exists(os.path.join(root, 'benchmarks', fm.get('rubric', 'x'))):
        print('judged case rubric missing:', f); ok = False
STATUS = {'proposed', 'approved', 'executed', 'verified', 'failed', 'superseded'}
for c in load_experiments(root + '/benchmarks/experiments'):
    if c.get('status') not in STATUS:
        print('bad experiment card:', c.get('_file')); ok = False
sys.exit(0 if ok else 1)
EOF
